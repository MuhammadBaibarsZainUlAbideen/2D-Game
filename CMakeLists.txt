cmake_minimum_required(VERSION 3.16)
project(CinematicSoldier LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(LOCAL_RAYLIB "${CMAKE_CURRENT_SOURCE_DIR}/build/_deps/raylib-build/raylib")
if(WIN32 AND EXISTS "${LOCAL_RAYLIB}/libraylib.a" AND EXISTS "${LOCAL_RAYLIB}/include/raylib.h")
add_library(raylib STATIC IMPORTED)
set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${LOCAL_RAYLIB}/libraylib.a" INTERFACE_INCLUDE_DIRECTORIES "${LOCAL_RAYLIB}/include" INTERFACE_LINK_LIBRARIES "winmm;gdi32;opengl32")
else()
set(FETCHCONTENT_QUIET FALSE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(raylib)
endif()
add_executable(CinematicSoldier main.cpp)
target_link_libraries(CinematicSoldier PRIVATE raylib)
include(CTest)
if(BUILD_TESTING)
add_executable(TunnelChecks tunnel_tests.cpp)
target_link_libraries(TunnelChecks PRIVATE raylib)
add_test(NAME TunnelPhysics COMMAND TunnelChecks)
endif()
